; Вектор сброса (Unified Entry Point)
; Адрес 0x00
00: D 1 0       JMP 0x10        ; Безусловный переход на начало программы

; ---------------------------------------------------------
; Основная программа (Program Space)
; ---------------------------------------------------------
START:
; Инициализация X = 0xF (Базовый старший ниббл для области MMIO F0-FF)
10: 1 F         LDI 0xF
12: 2 A         MOV X, A        ; Опкод 2A: D=1 (запись из A), RRR=010 (Регистр X). 1010b = A.

WAIT_KEY:
; Опрос регистра статуса клавиатуры (KBD_STAT, адрес 0xF4)
14: 1 4         LDI 0x4
16: 2 B         MOV Y, A        ; Опкод 2B: D=1, RRR=011 (Регистр Y). 1011b = B.
18: 3           LDR             ; A = Mem[F:4]. Бит 0 равен 1, если клавиша нажата.
19: 7 0         AND A           ; Неочевидная операция: "AND Reg" выполняет A = A & Reg.
                                ; В данном случае A = A & A. Сама математика ничего не меняет, 
                                ; но операция проходит через ALU и обновляет флаг Z (Zero).
1B: B 1 4       JZ 0x14         ; Если Z=1 (A равно нулю, клавиша не нажата), прыгаем в начало цикла 0x14.

; Клавиша нажата. Считываем скан-код (KBD_CODE, адрес 0xF5)
1E: 1 5         LDI 0x5
20: 2 B         MOV Y, A
22: 3           LDR             ; A = Mem[F:5]
23: 2 9         MOV B, A        ; Сохраняем скан-код в регистр B (Опкод 29: D=1, RRR=001)

; Выводим скан-код на дисплей DISP_0 (адрес 0xF0)
25: 1 0         LDI 0x0
27: 2 B         MOV Y, A        ; Адрес установлен на 0xF0
29: 2 1         MOV A, B        ; Восстанавливаем скан-код в A (Опкод 21: D=0, RRR=001)
2B: 4           STR             ; Mem[F:0] = A

; Включаем динамик (AUDIO, адрес 0xF6)
2C: 1 6         LDI 0x6
2E: 2 B         MOV Y, A        
30: 1 1         LDI 0x1         ; Загружаем 1 (бит 0 = On)
32: 4           STR             ; Mem[F:6] = 1

; Считываем случайное число (RNG, адрес 0xF7)
33: 1 7         LDI 0x7
35: 2 B         MOV Y, A
37: 3           LDR             ; A = Mem[F:7]
38: 2 9         MOV B, A        ; Прячем случайное число в B

; Выводим случайное число на дисплей DISP_3 (адрес 0xF3)
3A: 1 3         LDI 0x3
3C: 2 B         MOV Y, A
3E: 2 1         MOV A, B        
40: 4           STR             ; Mem[F:3] = A

; Выключаем динамик (AUDIO, адрес 0xF6)
; В реальном железе здесь понадобился бы цикл задержки, но в эмуляторе
; само выполнение инструкций создает достаточную визуально-звуковую паузу.
41: 1 6         LDI 0x6
43: 2 B         MOV Y, A
45: 1 0         LDI 0x0
47: 4           STR             ; Mem[F:6] = 0

WAIT_RELEASE:
; Ожидание отпускания клавиши (Защита от дребезга / множественного срабатывания)
48: 1 4         LDI 0x4
4A: 2 B         MOV Y, A
4C: 3           LDR             ; A = KBD_STAT
4D: 7 0         AND A           ; Обновляем флаг Z
4F: B 1 4       JZ 0x14         ; Если Z=1 (кнопка отпущена), прыгаем на WAIT_KEY (0x14)
52: D 4 8       JMP 0x48        ; Неочевидная архитектурная особенность: в ISA NC-1 нет инструкции 
                                ; JNZ (Jump if Not Zero). Поэтому для ветвления по "НЕ НУЛЮ" 
                                ; мы ставим JZ на выход из цикла, а следом безусловный JMP на начало цикла (0x48).
